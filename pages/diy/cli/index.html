// YOUTUBE DOWNLOAD PRO (YT-DLP)

Download and add Homebrew to your path

/usr/bin/ruby -e "$(curl -fsSL https://raw.githubusercontent.com/Homebrew/install/master/install)"

echo 'eval "$(/opt/homebrew/bin/brew shellenv)"' >> /Users/yourUserNameHere/.zprofile

eval "$(/opt/homebrew/bin/brew shellenv)"

install youtube download pro as well as ffmpeg (an open-source software project that provides a suite of libraries and programs for handling video, audio, and other multimedia files and streams)

brew install ffmpeg yt-dlp

brew update

download an audio file

yt-dlp -o "%(title)s.%(ext)s" -f bestaudio --ignore-errors --extract-audio --audio-format mp3 --audio-quality 0 --playlist-end 1 "https://www.youtube.com/watch?v=ZvIzmTiYoPo&list=RDZvIzmTiYoPo&start_radio=1"
